        document.getElementById('tPrev').addEventListener('click', () => go(current - 1, true));
        document.getElementById('tNext').addEventListener('click', () => go(current + 1, true));
        const tViewport = document.querySelector('.t-viewport');
        tViewport.addEventListener('mouseenter', () => clearInterval(timer));
        tViewport.addEventListener('mouseleave', restart);
        go(0); restart();

        /* ---------- Appointment form → WhatsApp ---------- */
        const form = document.getElementById('apptForm');
        const seg = document.getElementById('consultSeg');
        let consultType = 'In-Clinic';

        seg.querySelectorAll('button').forEach(btn => {
            btn.addEventListener('click', () => {
                seg.querySelectorAll('button').forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                consultType = btn.dataset.value;
            });
        });

        /* Date: min = today */
        const dateInput = document.getElementById('fDate');
        dateInput.min = new Date().toISOString().split('T')[0];

        function setErr(input, on) { input.closest('.field').classList.toggle('error', on); }

        form.addEventListener('submit', e => {
            e.preventDefault();
            const name = document.getElementById('fName').value.trim();
            const phone = document.getElementById('fPhone').value.replace(/\D/g, '');
            const date = dateInput.value;
            const slot = document.getElementById('fSlot').value;
            const msg = document.getElementById('fMsg').value.trim();

            let ok = true;
            setErr(document.getElementById('fName'), !name); ok = ok && !!name;
            const phoneOk = phone.length >= 10;
            setErr(document.getElementById('fPhone'), !phoneOk); ok = ok && phoneOk;
            setErr(dateInput, !date); ok = ok && !!date;
            setErr(document.getElementById('fSlot'), !slot); ok = ok && !!slot;
            if (!ok) return;

            const pretty = date ? new Date(date + 'T00:00').toLocaleDateString('en-IN', { weekday: 'short', day: 'numeric', month: 'short', year: 'numeric' }) : '';

            const text =
                `Namaste 🙏 New appointment request — Dr. Smrithi's Homeopathy Clinic\n\n` +
                `👤 Name: ${name}\n` +
                `📞 Phone: ${phone}\n` +
                `🩺 Consultation: ${consultType}\n` +
                `📅 Date: ${pretty}\n` +
                `🕐 Time: ${slot}` +
                (msg ? `\n💬 Message: ${msg}` : '') +
                `\n\n(_sent from the clinic website)`;

            const btn = document.getElementById('submitBtn');
            const original = btn.innerHTML;
            btn.innerHTML = 'Opening WhatsApp…';
            btn.disabled = true;

            window.open('https://wa.me/917338822967?text=' + encodeURIComponent(text), '_blank');

            setTimeout(() => { btn.innerHTML = original; btn.disabled = false; form.reset(); }, 3200);
        });

        /* Clear errors while typing */
        form.querySelectorAll('input, select').forEach(el =>
            el.addEventListener('input', () => setErr(el, false)));

        /* ---------- Year ---------- */
        document.getElementById('year').textContent = new Date().getFullYear();
